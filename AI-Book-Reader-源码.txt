  var lines=text.split('\n');
  var pats=[/^(Chapter|Ch\.?)\s+([IVXLCDM]+|\d+|[A-Z][a-z]+)\b.*$/i,/^(Book|Part|Volume|Vol\.?|Section|Act|Scene)\s+([IVXLCDM]+|\d+|[A-Z][a-z]+)\b.*$/i,/^CHAPTER\s+[IVXLCDM\d]+.*$/,/^[IVXLCDM]{2,}(\.|\s|$).*$/,/^(Prologue|Epilogue|Introduction|Preface|Foreword|Appendix|Index)\b.*$/i];
  var idx=[];
  for(var i=0;i<lines.length;i++){var line=lines[i].trim();if(!line||line.length>100)continue;for(var pi=0;pi<pats.length;pi++){if(pats[pi].test(line)){var prev=lines[i-1]?lines[i-1].trim():'',next=lines[i+1]?lines[i+1].trim():'';if(line.length<80&&(prev===''||i===0)&&(next===''||next.length>15)){idx.push({lineIdx:i,title:line});break}}}}
  var filt=[];for(var fi=0;fi<idx.length;fi++){if(filt.length===0||idx[fi].lineIdx-filt[filt.length-1].lineIdx>3)filt.push(idx[fi])}
  if(filt.length<2)return[{title:'Full Text',content:text}];
  var chs=[];
  for(var ci=0;ci<filt.length;ci++){var c=filt[ci],next=filt[ci+1];var start=0;for(var j=0;j<c.lineIdx;j++)start+=lines[j].length+1;var end=next?(function(){var e=0;for(var k=0;k<next.lineIdx;k++)e+=lines[k].length+1;return e})():text.length;var content=text.slice(start,end).trim();if(content.length>50)chs.push({title:c.title,content:content})}
  return chs;
}
function parseBook(file,onProg){
  var n=file.name.toLowerCase();var fmt=n.indexOf('.epub')>=0?'epub':n.indexOf('.pdf')>=0?'pdf':n.indexOf('.docx')>=0?'docx':'txt';
  if(onProg)onProg({current:0,total:100,stage:'Reading file'});
  var text='',title=file.name.replace(/\.[^.]+$/,'');
  var p;
  if(fmt==='txt')p=file.text();
  else if(fmt==='epub')p=parseEPUB(file,onProg);
  else if(fmt==='pdf')p=parsePDF(file,onProg);
  else if(fmt==='docx')p=parseDOCX(file,onProg);
  return p.then(function(t){text=t}).catch(function(e){console.error(e);return file.text().catch(function(){throw new Error('Parse failed')})}).then(function(){
    if(onProg)onProg({current:60,total:100,stage:'Detecting chapters'});
    var clean=text.replace(/\r\n/g,'\n').replace(/\r/g,'\n').replace(/\t/g,' ').replace(/\u00A0/g,' ').replace(/[ \t]+/g,' ').replace(/\n{3,}/g,'\n\n').trim();
    var chs=detectChapters(clean);
    var fl=clean.split('\n').slice(0,5).filter(function(l){return l.trim()});
    if(fl.length>0&&fl[0].length<80&&!/^(chapter|book|part)/i.test(fl[0]))title=fl[0].trim();
    if(onProg)onProg({current:100,total:100,stage:'Done'});
    return{title:title,format:fmt,chapters:chs.map(function(c){return c.content}),chapterTitles:chs.map(function(c){return c.title}),fileSize:file.size};
  });
}
function parseEPUB(file,onProg){
  return JSZip.loadAsync(file.arrayBuffer()).then(function(zip){
    if(onProg)onProg({current:20,total:100,stage:'Parsing EPUB'});
    return zip.file('META-INF/container.xml').async('string').catch(function(){return null}).then(function(cxml){
      var opf='content.opf';if(cxml){var m=cxml.match(/full-path="([^"]+)"/);if(m)opf=m[1]}
      var dir=opf.indexOf('/')>=0?opf.replace(/\/[^/]+$/,'/'):'';
      return zip.file(opf).async('string').catch(function(){return null}).then(function(oxml){
        if(!oxml)throw new Error('OPF not found');
        var sm=oxml.match(/<spine[^>]*>([\s\S]*?)<\/spine>/i);var refs=[];
        if(sm){var rs=sm[1].match(/<itemref[^>]*idref="([^"]+)"/gi)||[];for(var ri=0;ri<rs.length;ri++){var m2=rs[ri].match(/idref="([^"]+)"/i);if(m2)refs.push(m2[1])}}
        var mm=oxml.match(/<manifest[^>]*>([\s\S]*?)<\/manifest>/i);var map={};
        if(mm){var items=mm[1].match(/<item[^>]*>/gi)||[];for(var ii=0;ii<items.length;ii++){var a=items[ii].match(/id="([^"]+)"/i),b=items[ii].match(/href="([^"]+)"/i);if(a&&b)map[a[1]]=b[1]}}
        if(onProg)onProg({current:40,total:100,stage:'Extracting text'});
        var arr=[];var chain=Promise.resolve();
        refs.forEach(function(ref){chain=chain.then(function(){var href=map[ref];if(!href)return;var entry=zip.file(dir+href)||zip.file(decodeURIComponent(dir+href));if(!entry)return;return entry.async('string').then(function(html){var t=html.replace(/<script[\s\S]*?<\/script>/gi,'').replace(/<style[\s\S]*?<\/style>/gi,'').replace(/<\/(p|div|h[1-6]|li|br|tr|blockquote)>/gi,'\n').replace(/<br\s*\/?>/gi,'\n').replace(/<[^>]+>/g,'').replace(/&nbsp;/g,' ').replace(/&amp;/g,'&').replace(/&lt;/g,'<').replace(/&gt;/g,'>').replace(/&quot;/g,'"').replace(/&#39;/g,"'");if(t.trim().length>30)arr.push(t.trim())})})});
        return chain.then(function(){return arr.join('\n\n')});
      });
    });
  });
}

}
